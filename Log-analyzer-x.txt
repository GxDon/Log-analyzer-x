#!/usr/bin/env python3

import os
from collections import Counter
from datetime import datetime


LOG_FILE = "logs/monitor.log"
REPORT_FILE = "reports/report.txt"

REPEAT_THRESHOLD = 5


def preparar_directorios():
    os.makedirs("logs", exist_ok=True)
    os.makedirs("reports", exist_ok=True)


def leer_logs():
    if not os.path.isfile(LOG_FILE):
        print(f"[!] No existe el archivo: {LOG_FILE}")
        return []

    try:
        with open(LOG_FILE, "r", encoding="utf-8", errors="replace") as file:
            return [
                line.strip()
                for line in file
                if line.strip()
            ]

    except OSError as error:
        print(f"[!] Error al leer el log: {error}")
        return []


def extraer_evento(linea):
    if "]" in linea:
        try:
            return linea.split("]", 1)[1].strip()
        except IndexError:
            return linea.strip()

    return linea.strip()


def analizar_logs(lineas):
    eventos = []

    for linea in lineas:
        evento = extraer_evento(linea)

        if evento:
            eventos.append(evento)

    contador = Counter(eventos)

    alertas = []

    for evento, cantidad in contador.items():

        if cantidad >= REPEAT_THRESHOLD:
            alertas.append(
                f"REPETICIÓN: '{evento}' apareció {cantidad} veces."
            )

        evento_lower = evento.lower()

        if "fallido" in evento_lower:
            alertas.append(
                f"ADVERTENCIA: se detectó un acceso fallido: '{evento}'."
            )

        if "sospechosa" in evento_lower:
            alertas.append(
                f"CRÍTICO: actividad sospechosa detectada: '{evento}'."
            )

    return contador, alertas


def generar_reporte(contador, alertas):
    fecha = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    try:
        with open(REPORT_FILE, "w", encoding="utf-8") as report:

            report.write("LOG ANALYZER X\n")
            report.write("=" * 50 + "\n")
            report.write(f"Fecha del análisis: {fecha}\n")
            report.write(f"Archivo analizado: {LOG_FILE}\n\n")

            report.write("EVENTOS DETECTADOS\n")
            report.write("-" * 50 + "\n")

            if contador:
                for evento, cantidad in contador.most_common():
                    report.write(
                        f"{cantidad:>4} | {evento}\n"
                    )
            else:
                report.write("No se encontraron eventos.\n")

            report.write("\nALERTAS\n")
            report.write("-" * 50 + "\n")

            if alertas:
                for alerta in alertas:
                    report.write(f"- {alerta}\n")
            else:
                report.write("No se detectaron alertas.\n")

        return True

    except OSError as error:
        print(f"[!] Error al generar el reporte: {error}")
        return False


def main():
    preparar_directorios()

    print("=" * 50)
    print("          Log-analyzer-x")
    print("=" * 50)

    lineas = leer_logs()

    if not lineas:
        print("[!] No hay información para analizar.")
        return

    contador, alertas = analizar_logs(lineas)

    print(f"[+] Líneas analizadas: {len(lineas)}")
    print(f"[+] Eventos diferentes: {len(contador)}")
    print(f"[+] Alertas: {len(alertas)}")

    if generar_reporte(contador, alertas):
        print(f"[+] Reporte generado: {REPORT_FILE}")


if __name__ == "__main__":
    main()
